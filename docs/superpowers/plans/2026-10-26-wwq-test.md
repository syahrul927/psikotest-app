# WWQ (Woodworth's Questionnaire) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add WWQ as a fourth test type: admin creates invitations, guest testers answer 75 YA/TIDAK questions via invitation link, server auto-scores into 8 aspects, admin reviews result (radar + table) on dashboard.

**Architecture:** Clone the PAPI-Kostick vertical slice end-to-end (per-test table family, dual tRPC routers public/protected, guest pages with access-gate provider, dashboard result page with context provider). All scoring logic is a pure module (`src/lib/wwq-utils.ts`) reverse-engineered from `docs/WWQ.xlsx` and verified against its cached values. Question text lives in DB (seeded from `docs/wwq-data/wwq-questions.json`); scoring key lives in code (same split as PAPI).

**Tech Stack:** Next.js 15 App Router, tRPC v11, Prisma + PostgreSQL, TanStack Query, Zod, shadcn/ui + Tailwind, recharts, framer-motion, sonner. Package manager: **yarn**.

**Spec:** `docs/superpowers/specs/2026-10-26-wwq-test-design.md`

```ts
  | "Kecendrungan Pathologis"
  | "Gangguan Pathologis";

// Column-K rule from the sheet, applied to a weighted value.
export function wwqCategory(weighted: number): WwqCategory {
  if (weighted >= 181) return "Gangguan Pathologis";
  if (weighted >= 120) return "Kecendrungan Pathologis";
  return "Normal";
}

export interface WwqAspectScore {
  aspectNo: WwqAspectNo;
  aspect: string;
  rawScore: number;
  acuan: number;
  weighted: number;
  percent: number;
  category: WwqCategory;
}

export interface WwqScoreResult {
  aspects: WwqAspectScore[];
  totalWeighted: number;
  overallValue: number; // totalWeighted / 8 ("Total Keseluruhan")
  category: WwqCategory;
}

/**
 * @param answers 75 booleans indexed by question order (1..75). true = YA.
 */
export function calculateWWQScores(answers: Map<number, boolean>): WwqScoreResult {
  if (answers.size !== WWQ_TOTAL_QUESTIONS) {
    throw new Error(`Jawaban harus ${WWQ_TOTAL_QUESTIONS} soal, diterima: ${answers.size}`);
  }

  const raw = new Map<WwqAspectNo, number>(WWQ_ASPECTS.map((a) => [a.no, 0]));

  for (const [questionOrder, aspectNo, inverted] of WWQ_KEY) {
    const answer = answers.get(questionOrder);
    if (answer === undefined) {
      throw new Error(`Jawaban soal ke-${questionOrder} belum tersedia`);
    }
    const scored = inverted ? answer === false : answer === true;
    if (scored) raw.set(aspectNo, (raw.get(aspectNo) ?? 0) + 1);
  }

  const aspects: WwqAspectScore[] = WWQ_ASPECTS.map((meta) => {
    const rawScore = raw.get(meta.no) ?? 0;
    const weighted = rawScore * meta.acuan;
    return {
      aspectNo: meta.no,
      aspect: meta.name,
      rawScore,
      acuan: meta.acuan,
      weighted,
      percent: 0, // filled below
      category: wwqCategory(weighted),
    };
  });

  const totalWeighted = aspects.reduce((s, a) => s + a.weighted, 0);
  for (const a of aspects) {
    // guard: all-zero answers -> 0% (sheet shows Normal, not NaN)
    a.percent = totalWeighted === 0 ? 0 : (a.weighted * 100) / totalWeighted;
  }

  const overallValue = totalWeighted / 8;

  return {
    aspects,
    totalWeighted,
    overallValue,
    category: wwqCategory(overallValue),
  };
}
```

```ts
// Seeds WwqQuestion from docs/wwq-data/wwq-questions.json (extracted from docs/WWQ.xlsx).
// Idempotent: upserts by `order`. Run manually: npx -y tsx prisma/seed-wwq.ts
import { PrismaClient } from "@prisma/client";
import fs from "node:fs";
import path from "node:path";

const prisma = new PrismaClient();

interface WwqQuestionJson {
  no: number;
  text: string;
}

async function main() {
  const file = path.join(process.cwd(), "docs", "wwq-data", "wwq-questions.json");
  const questions = JSON.parse(fs.readFileSync(file, "utf-8")) as WwqQuestionJson[];
  if (questions.length !== 75) throw new Error(`Expected 75 questions, got ${questions.length}`);

  for (const q of questions) {
    await prisma.wwqQuestion.upsert({
      where: { order: q.no },
      create: { order: q.no, description: q.text },
      update: { description: q.text },
    });
  }
  const count = await prisma.wwqQuestion.count();
  console.log(`Seeded wwq questions: ${count}`);
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(() => prisma.$disconnect());
```

- [ ] **Step 2: `public/index.ts`** — clone of `papi-kostick-test-router/public/index.ts` with WWQ scoring. Key differences from PAPI: question id is `order` (PAPI parses cuid as Number); boolean answers; server-side status guard; nested result create carries totals.

```ts
import {
  calculateWWQScores,
  WWQ_TOTAL_QUESTIONS,
} from "@/lib/wwq-utils";
import { createTRPCRouter, publicProcedure } from "@/server/api/trpc";
import type { Prisma } from "@prisma/client";
import { TRPCError } from "@trpc/server";
import { SubmitAnswerWwqRequest } from "../schema";

export const wwqTestPublicRouter = createTRPCRouter({
  findAllQuestions: publicProcedure.query(async ({ ctx }) => {
    const questions = await ctx.db.wwqQuestion.findMany({
      select: { id: true, order: true, description: true },
      orderBy: { order: "asc" },
    });
    return questions;
  }),

  submitAnswers: publicProcedure
    .input(SubmitAnswerWwqRequest)
    .mutation(async ({ ctx, input }) => {
      if (input.data.length !== WWQ_TOTAL_QUESTIONS) {
        throw new TRPCError({
          code: "BAD_REQUEST",
          message: `Invalid input: the total number of answers must be ${WWQ_TOTAL_QUESTIONS}`,
        });
      }
      if (new Set(input.data.map((d) => d.questionId)).size !== WWQ_TOTAL_QUESTIONS) {
        throw new TRPCError({
          code: "BAD_REQUEST",
          message: "Invalid input: duplicate or missing question ids",
        });
      }

      const invitation = await ctx.db.wwqInvitation.findUnique({
        where: { id: input.invitationId },
      });
      if (!invitation || invitation.status !== "ONPROGRESS") {
        throw new TRPCError({
          code: "BAD_REQUEST",
          message: "Invitation is not in progress",
        });
      }

      const questions = await ctx.db.wwqQuestion.findMany({
        select: { id: true, order: true },
      });
      const orderById = new Map(questions.map((q) => [q.id, q.order]));

      const answers = new Map<number, boolean>();
      for (const d of input.data) {
        // client sends DB question ids (cuid); scoring key is by order
        const order = orderById.get(d.questionId);
        if (order === undefined) {
          throw new TRPCError({ code: "BAD_REQUEST", message: "Unknown question id" });
        }
        answers.set(order, d.answer);
      }

      const scores = calculateWWQScores(answers);

      const resultDetails: Prisma.WwqResultDetailCreateWithoutResultInput[] =
        scores.aspects.map((a) => ({
          aspectNo: a.aspectNo,
          aspect: a.aspect,
          rawScore: a.rawScore,
          acuan: a.acuan,
          weighted: a.weighted,
          percent: a.percent,
          category: a.category,
        }));

      await ctx.db.$transaction([
        ctx.db.wwqAnswer.createMany({
          data: input.data.map((i) => ({
            invitationId: input.invitationId,
            questionId: i.questionId,
            answer: i.answer,
          })),
        }),
        ctx.db.wwqInvitation.update({
          where: { id: input.invitationId },
          data: {
            WwqResult: {
              create: {
                totalWeighted: scores.totalWeighted,
                overallValue: scores.overallValue,
                category: scores.category,
                completedAt: new Date(),
                WwqResultDetail: { createMany: { data: resultDetails } },
              },
            },
            status: "DONE",
          },
        }),
      ]);

      return { success: true };
    }),
});
```

Client sends the DB `id` string (matches `schema.ts` above). `WwqAnswer.questionId` references `WwqQuestion.id` (cuid).
